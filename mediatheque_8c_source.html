43void lireLigne(char *dest, int taille)
44{
45 if (fgets(dest, taille, stdin) == NULL)
46 {
47 /* Fin du flux d'entrée (EOF) : fgets n'a rien écrit dans dest, on y
48 met une chaîne vide pour ne pas lire un contenu non initialisé. */
49 dest[0] = '\0';
50 return;
51 }
52 size_t fin = strlen(dest);
53 if (fin > 0 && dest[fin - 1] == '\n')
54 {
55 /* fgets garde le '\n' final : on le retire pour ne stocker que le texte. */
56 dest[fin - 1] = '\0';
57 }
58 else
59 {
60 /* La ligne saisie était plus longue que le buffer : fgets n'a pas pu
61 lire le '\n' de fin de ligne, il reste donc dans le flux. On le
62 vide ici pour ne pas perturber la prochaine lecture. */
63 viderBuffer();
64 }
65}

80int lireEntier(void)
81{
82 char saisie[32];
83 char *fin;
84 long valeur;
85 int chiffreLu;
86
87 while (1)
88 {
89 if (fgets(saisie, sizeof saisie, stdin) == NULL)
90 {
91 /* Fin du flux (EOF) : on renvoie 0, ce qui fait quitter le menu
92 proprement au lieu de redemander indéfiniment. */
93 return 0;
94 }
95
96 size_t longueur = strlen(saisie);
97 if (longueur > 0 && saisie[longueur - 1] != '\n')
98 {
99 /* Ligne trop longue pour un entier : on jette le reste de la
100 ligne et on considère la saisie comme invalide. */
101 viderBuffer();
102 printf("Saisie invalide, entrez un nombre entier : ");
103 continue;
104 }
105
106 /* strtol convertit le début de la chaîne et place fin juste après le
107 dernier chiffre lu : la saisie n'est valide que si au moins un
108 chiffre a été lu (fin != saisie) et s'il ne reste ensuite que des
109 espaces ou le '\n' final. errno signale un dépassement de long.
110 Le test fin != saisie est mémorisé AVANT de sauter les espaces :
111 sinon une ligne vide (Entrée seule) ou faite d'espaces ferait
112 avancer fin et serait acceptée comme le nombre 0. */
113 errno = 0;
114 valeur = strtol(saisie, &fin, 10);
115 chiffreLu = (fin != saisie);
116 while (isspace((unsigned char)*fin))
117 {
118 fin++;
119 }
120
121 if (chiffreLu && *fin == '\0' && errno == 0
122 && valeur >= INT_MIN && valeur <= INT_MAX)
123 {
124 return (int)valeur;
125 }
126
127 printf("Saisie invalide, entrez un nombre entier : ");
128 }
129}

492 if (indice == -1)
