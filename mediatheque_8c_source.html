43void lireLigne(char *dest, int taille)
44{
45 if (fgets(dest, taille, stdin) == NULL)
46 {
47 /* Fin du flux d'entrée (EOF) : fgets n'a rien écrit dans dest, on y
48 met une chaîne vide pour ne pas lire un contenu non initialisé. */
49 dest[0] = '\0';
50 return;
51 }
52 size_t fin = strlen(dest);
53 if (fin > 0 && dest[fin - 1] == '\n')
54 {
55 /* fgets garde le '\n' final : on le retire pour ne stocker que le texte. */
56 dest[fin - 1] = '\0';
57 }
58 else
59 {
60 /* La ligne saisie était plus longue que le buffer : fgets n'a pas pu
61 lire le '\n' de fin de ligne, il reste donc dans le flux. On le
62 vide ici pour ne pas perturber la prochaine lecture. */
63 viderBuffer();
64 }
65}

80int lireEntier(void)
81{
82 char saisie[32];
83 char *fin;
84 long valeur;
85
86 while (1)
87 {
88 if (fgets(saisie, sizeof saisie, stdin) == NULL)
89 {
90 /* Fin du flux (EOF) : on renvoie 0, ce qui fait quitter le menu
91 proprement au lieu de redemander indéfiniment. */
92 return 0;
93 }
94
95 size_t longueur = strlen(saisie);
96 if (longueur > 0 && saisie[longueur - 1] != '\n')
97 {
98 /* Ligne trop longue pour un entier : on jette le reste de la
99 ligne et on considère la saisie comme invalide. */
100 viderBuffer();
101 printf("Saisie invalide, entrez un nombre entier : ");
102 continue;
103 }
104
105 /* strtol convertit le début de la chaîne et place fin juste après le
106 dernier chiffre lu : la saisie n'est valide que si au moins un
107 chiffre a été lu (fin != saisie) et s'il ne reste ensuite que des
108 espaces ou le '\n' final. errno signale un dépassement de long. */
109 errno = 0;
110 valeur = strtol(saisie, &fin, 10);
111 while (isspace((unsigned char)*fin))
112 {
113 fin++;
114 }
115
116 if (fin != saisie && *fin == '\0' && errno == 0
117 && valeur >= INT_MIN && valeur <= INT_MAX)
118 {
119 return (int)valeur;
120 }
121
122 printf("Saisie invalide, entrez un nombre entier : ");
123 }
124}
